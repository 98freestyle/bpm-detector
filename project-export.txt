=== README.md ===

# BPM Analyzer

A modern web application for detecting BPM (beats per minute) in audio files.

## Architecture

- **Frontend**: React + Vite + TailwindCSS
- **Backend**: Python Flask + librosa
- **Processing**: Server-side audio analysis for high accuracy

## Development Setup

### Frontend
```bash
cd frontend
npm install
npm run dev
```

### Backend  
```bash
cd backend
python -m venv venv
venv\Scripts\activate  # Windows
pip install flask librosa flask-cors
python app.py
```

## Features
- Drag & drop file upload
- High-accuracy BPM detection for electronic music
- Modern responsive UI
- Fast server-side processing

=== backend/app.py ===

from flask import Flask, request, jsonify
from flask_cors import CORS
import essentia
import essentia.standard as es
import tempfile
import os

app = Flask(__name__)
CORS(app)

@app.route('/analyze', methods=['POST'])
def analyze_bpm():
    if 'audio' not in request.files:
        return jsonify({'error': 'No audio file'}), 400
    
    file = request.files['audio']
    temp_path = None
    
    try:
        with tempfile.NamedTemporaryFile(delete=False, suffix='.wav') as tmp:
            temp_path = tmp.name
            file.save(tmp.name)
        
        # Load audio with Essentia
        loader = es.MonoLoader(filename=temp_path)
        audio = loader()
        
        # Use RhythmExtractor2013 - the actual algorithm
        rhythm_extractor = es.RhythmExtractor2013()
        bpm, beats, beats_confidence, _, beats_intervals = rhythm_extractor(audio)
        
        return jsonify({'bpm': int(round(bpm))})
        
    except Exception as e:
        print(f"Error: {e}")
        return jsonify({'error': str(e)}), 500
        
    finally:
        if temp_path and os.path.exists(temp_path):
            try:
                os.unlink(temp_path)
            except:
                pass

if __name__ == '__main__':
    app.run(debug=True)

import { useState } from "react";
import { motion } from "framer-motion";
import { Upload } from "lucide-react"; // lightweight icon library

function Dropzone({ onFile }) {
    const [dragOver, setDragOver] = useState(false);

    function handleDrop(e) {
        e.preventDefault();
        setDragOver(false);
        const file = e.dataTransfer.files[0];
        onFile(file);
    }

    function handleBrowse(e) {
        const file = e.target.files[0];
        onFile(file);
    }

    return (
        <motion.label
            htmlFor="file-upload"
            className={`relative flex flex-col items-center justify-center p-12 rounded-2xl cursor-pointer transition-all
        border-2 border-dashed ${dragOver
                    ? "border-brand bg-brand/10 shadow-lg scale-105"
                    : "border-gray-600 hover:border-brand hover:bg-gray-900/60"
                }`}
            onDragOver={(e) => {
                e.preventDefault();
                setDragOver(true);
            }}
            onDragLeave={() => setDragOver(false)}
            onDrop={handleDrop}
            whileHover={{ scale: 1.02 }}
            whileTap={{ scale: 0.98 }}
        >
            {/* Icon */}
            <Upload className="w-10 h-10 text-brand mb-4" />

            {/* Text */}
            <p className="text-gray-300 text-lg">
                <span className="font-semibold text-brand">Click to browse</span> or
                drag & drop your audio file here
            </p>

            {/* Hidden input */}
            <input
                id="file-upload"
                type="file"
                accept="audio/*"
                onChange={handleBrowse}
                className="hidden"
            />
        </motion.label>
    );
}

export default Dropzone;

=== frontend/src/components/Hero.jsx ===

import { useState } from "react";
import Dropzone from "./Dropzone";
import Loader from "./Loader";
import Result from "./Result";
import { motion } from "framer-motion";
import { useBPMDetector } from "../utils/useBPMDetector";

function Hero() {
    // State for file analysis results
    const [fileName, setFileName] = useState(null);
    const [bpm, setBpm] = useState(null);
    const [error, setError] = useState(null);
    const [loading, setLoading] = useState(false);

    // Load BPM detector
    const bpmDetector = useBPMDetector();

    /**
     * Handles audio file upload + BPM analysis
     */
    async function handleFile(file) {
        if (!file) return;

        setError(null);
        setBpm(null);
        setLoading(true);

        // Validate file type
        if (!file.type.startsWith("audio/")) {
            setError("Only audio files are supported.");
            setLoading(false);
            return;
        }

        setFileName(file.name);

        // Validate file size (150MB limit)
        const MAX_FILE_SIZE = 150 * 1024 * 1024;
        if (file.size > MAX_FILE_SIZE) {
            setError("File too large. Max allowed is 150MB.");
            setLoading(false);
            return;
        }

        try {
            // Send file to backend API
            const result = await bpmDetector.detect(file);

            console.log("Backend result:", result);

            if (result && result.bpm) {
                setBpm(result.bpm);
            } else {
                setError("Could not detect BPM.");
            }

            setLoading(false);
        } catch (err) {
            console.error("BPM detection error:", err);
            setError("BPM detection failed.");
            setLoading(false);
        }
    }

    return (
        <motion.header
            className="flex-1 flex items-center justify-center text-center px-6 py-20 bg-gradient-to-r from-brand-dark to-brand-light"
            initial={{ opacity: 0, y: 60 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ duration: 0.8 }}
        >
            <div className="max-w-3xl w-full">
                {/* Hero headline */}
                <h2 className="text-5xl font-extrabold leading-tight">
                    Detect the <span className="text-accent">BPM</span> of your music instantly
                </h2>
                <p className="mt-6 text-lg text-gray-200">
                    Upload or drag & drop your track — get clean, accurate BPM analysis in seconds.
                </p>

                {/* Analyzer Card */}
                <div className="mt-10 w-full max-w-xl mx-auto bg-gray-900 p-8 rounded-2xl shadow-lg border border-gray-700">
                    <Dropzone onFile={handleFile} />
                    {loading && <Loader />}
                    {error && <p className="text-red-400 font-medium mt-4">{error}</p>}
                    <Result fileName={fileName} bpm={bpm} />
                </div>
            </div>
        </motion.header>
    );
}

export default Hero;

export function useBPMDetector() {
    return {
        detect: async (audioFile) => {
            const formData = new FormData();
            formData.append('audio', audioFile);

            const response = await fetch('http://127.0.0.1:5000/analyze', {
                method: 'POST',
                body: formData
            });

            if (!response.ok) {
                throw new Error('Backend analysis failed');
            }

            return await response.json();
        }
    };
}
